cmake_minimum_required(VERSION 3.22)
project(manyue_lite LANGUAGES C CXX)

# Zig's x86_64-windows-gnu driver is used for host numerical Vulkan tests when
# MSVC is unavailable. CMake detects Zig's clang front-end, but the standalone
# compiler files may still report AR-NOTFOUND; accept explicit Zig-compatible
# archive tools and bind all static target rules to them.
set(MANYUE_LITE_HOST_ARCHIVER "" CACHE FILEPATH "Host archiver, e.g. Zig ar wrapper")
set(MANYUE_LITE_HOST_RANLIB "" CACHE FILEPATH "Host ranlib, e.g. Zig ranlib wrapper")
if(MANYUE_LITE_HOST_ARCHIVER)
    if(NOT MANYUE_LITE_HOST_RANLIB)
        message(FATAL_ERROR "MANYUE_LITE_HOST_RANLIB is required with MANYUE_LITE_HOST_ARCHIVER")
    endif()
    set(CMAKE_AR "${MANYUE_LITE_HOST_ARCHIVER}")
    set(CMAKE_RANLIB "${MANYUE_LITE_HOST_RANLIB}")
    set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> qc <TARGET> <LINK_FLAGS> <OBJECTS>")
    set(CMAKE_C_ARCHIVE_APPEND "<CMAKE_AR> q <TARGET> <LINK_FLAGS> <OBJECTS>")
    set(CMAKE_C_ARCHIVE_FINISH "<CMAKE_RANLIB> <TARGET>")
    set(CMAKE_CXX_ARCHIVE_CREATE "<CMAKE_AR> qc <TARGET> <LINK_FLAGS> <OBJECTS>")
    set(CMAKE_CXX_ARCHIVE_APPEND "<CMAKE_AR> q <TARGET> <LINK_FLAGS> <OBJECTS>")
    set(CMAKE_CXX_ARCHIVE_FINISH "<CMAKE_RANLIB> <TARGET>")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(MANYUE_LITE_NCNN_ROOT "" CACHE PATH "Pinned ncnn SDK root")
set(MANYUE_LITE_NCNN_LIBRARY "" CACHE FILEPATH "Host or Android ncnn shared library")
set(MANYUE_LITE_NCNN_IMPLIB "" CACHE FILEPATH "Windows host ncnn import library")
set(MANYUE_LITE_NCNN_INCLUDE_DIR "" CACHE PATH "ncnn include directory")
set(MANYUE_LITE_WEBP_ROOT "" CACHE PATH "Pinned libwebp 1.5.0 source root")
option(MANYUE_LITE_BUILD_HOST_TESTS "Build the host numerical golden test runner" OFF)

if(ANDROID)
    if(NOT MANYUE_LITE_NCNN_LIBRARY)
        set(MANYUE_LITE_NCNN_LIBRARY "${MANYUE_LITE_NCNN_ROOT}/arm64-v8a/lib/libncnn.so")
    endif()
    if(NOT MANYUE_LITE_NCNN_INCLUDE_DIR)
        set(MANYUE_LITE_NCNN_INCLUDE_DIR "${MANYUE_LITE_NCNN_ROOT}/arm64-v8a/include")
    endif()
elseif(NOT MANYUE_LITE_NCNN_LIBRARY AND MANYUE_LITE_NCNN_ROOT)
    if(WIN32)
        set(MANYUE_LITE_NCNN_LIBRARY "${MANYUE_LITE_NCNN_ROOT}/bin/ncnn.dll")
        set(MANYUE_LITE_NCNN_IMPLIB "${MANYUE_LITE_NCNN_ROOT}/lib/ncnn.lib")
        set(MANYUE_LITE_NCNN_INCLUDE_DIR "${MANYUE_LITE_NCNN_ROOT}/include")
    else()
        set(MANYUE_LITE_NCNN_LIBRARY "${MANYUE_LITE_NCNN_ROOT}/lib/libncnn.so")
        set(MANYUE_LITE_NCNN_INCLUDE_DIR "${MANYUE_LITE_NCNN_ROOT}/include")
    endif()
endif()
if(NOT EXISTS "${MANYUE_LITE_NCNN_LIBRARY}")
    message(FATAL_ERROR "Set MANYUE_LITE_NCNN_LIBRARY to the ncnn library for this target")
endif()
if(NOT EXISTS "${MANYUE_LITE_NCNN_INCLUDE_DIR}/ncnn/net.h")
    message(FATAL_ERROR "Pinned ncnn headers are missing")
endif()
if(NOT EXISTS "${MANYUE_LITE_WEBP_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "MANYUE_LITE_WEBP_ROOT must point to libwebp 1.5.0 sources")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build bundled WebP statically" FORCE)
set(WEBP_BUILD_ANIM_UTILS OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_CWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_DWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_GIF2WEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_IMG2WEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_VWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_WEBPINFO OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_WEBPMUX OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_LIBWEBPMUX OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_WEBP_JS OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_FUZZTEST OFF CACHE BOOL "" FORCE)
set(WEBP_USE_THREAD OFF CACHE BOOL "" FORCE)
add_subdirectory("${MANYUE_LITE_WEBP_ROOT}" "${CMAKE_BINARY_DIR}/libwebp" EXCLUDE_FROM_ALL)

add_library(ncnn SHARED IMPORTED GLOBAL)
set_target_properties(ncnn PROPERTIES
    IMPORTED_LOCATION "${MANYUE_LITE_NCNN_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${MANYUE_LITE_NCNN_INCLUDE_DIR}")
if(WIN32 AND MANYUE_LITE_NCNN_IMPLIB)
    set_target_properties(ncnn PROPERTIES IMPORTED_IMPLIB "${MANYUE_LITE_NCNN_IMPLIB}")
endif()

add_library(manyue_lite_core STATIC
    manyue_lite_engine.cpp
    manyue_lite_reference.cpp)
target_include_directories(manyue_lite_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(manyue_lite_core PUBLIC ncnn webp)
if(MSVC)
    target_compile_options(manyue_lite_core PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(manyue_lite_core PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()

if(ANDROID)
    add_library(manyue_lite SHARED manyue_lite_jni.cpp)
    target_include_directories(manyue_lite PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(manyue_lite PRIVATE manyue_lite_core ncnn webp log android)
    target_compile_options(manyue_lite PRIVATE -Wall -Wextra -Werror)
    target_link_options(manyue_lite PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/manyue_lite.exports.map")
    target_compile_options(manyue_lite_core PRIVATE -fvisibility=hidden)
    set_target_properties(manyue_lite PROPERTIES OUTPUT_NAME manyue_lite)
elseif(MANYUE_LITE_BUILD_HOST_TESTS)
    add_executable(manyue_lite_reference_test tests/reference_test.cpp)
    target_link_libraries(manyue_lite_reference_test PRIVATE manyue_lite_core)
    if(MSVC)
        target_compile_options(manyue_lite_reference_test PRIVATE /W4 /WX /EHsc)
    else()
        target_compile_options(manyue_lite_reference_test PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    endif()

    add_executable(manyue_lite_host_test tests/manyue_lite_host_test.cpp)
    target_link_libraries(manyue_lite_host_test PRIVATE manyue_lite_core webp)
    if(MSVC)
        target_compile_options(manyue_lite_host_test PRIVATE /W4 /WX /EHsc)
    else()
        target_compile_options(manyue_lite_host_test PRIVATE -Wall -Wextra -Werror)
    endif()
endif()
